cmake_minimum_required(VERSION 3.20)
project(Denoise VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(NOT TARGET vv_core)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../shared/engine ${CMAKE_BINARY_DIR}/vv-engine)
endif()

# Research phase: CPU core + evaluation harness only (no host plug-in yet).
add_library(dn_core STATIC src/core/dn_pyramid.cpp src/core/dn_profile.cpp src/core/dn_starlet.cpp src/core/dn_spatial.cpp src/core/dn_temporal.cpp)
target_include_directories(dn_core PUBLIC src/core)
target_link_libraries(dn_core PUBLIC vv_core)
set_target_properties(dn_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(dn_eval STATIC src/eval/dn_synth.cpp src/eval/dn_metrics.cpp
  src/eval/dn_sequence.cpp src/eval/dn_sequence_metrics.cpp)
target_include_directories(dn_eval PUBLIC src/eval)
target_link_libraries(dn_eval PUBLIC dn_core)

if(NOT MSVC)
  target_compile_options(dn_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(dn_eval PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(dn_core_tests tests/core/test_main.cpp)
target_link_libraries(dn_core_tests PRIVATE dn_eval)
add_executable(dn_eval_tool tools/dn_eval.cpp)
target_link_libraries(dn_eval_tool PRIVATE dn_eval)
add_executable(dn_temporal_eval tools/dn_temporal_eval.cpp)
target_link_libraries(dn_temporal_eval PRIVATE dn_eval)
add_executable(dn_sequence_tool tools/dn_sequence.cpp)
target_link_libraries(dn_sequence_tool PRIVATE dn_eval)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE DN_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE DN_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT DN_REVISION)
  set(DN_REVISION unknown)
elseif(DN_DIRTY)
  string(APPEND DN_REVISION "-dirty")
endif()
file(GLOB_RECURSE DN_PROVENANCE_SOURCES CONFIGURE_DEPENDS
  ${CMAKE_CURRENT_SOURCE_DIR}/src/*.h ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp
  ${VV_ENGINE_DIR}/core/vv/*.h ${VV_ENGINE_DIR}/core/vv/*.cpp)
list(APPEND DN_PROVENANCE_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/tools/dn_sequence.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt ${VV_ENGINE_DIR}/io/pfm.h)
list(SORT DN_PROVENANCE_SOURCES)
set(DN_SOURCE_RECORD "")
foreach(source IN LISTS DN_PROVENANCE_SOURCES)
  file(SHA256 ${source} digest)
  file(RELATIVE_PATH label ${CMAKE_CURRENT_SOURCE_DIR}/../.. ${source})
  string(APPEND DN_SOURCE_RECORD "${label}:${digest}\n")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${DN_PROVENANCE_SOURCES})
string(SHA256 DN_SOURCE_DIGEST "${DN_SOURCE_RECORD}")
target_compile_definitions(dn_sequence_tool PRIVATE DN_SOURCE_REVISION="${DN_REVISION}"
  DN_SOURCE_DIGEST="${DN_SOURCE_DIGEST}")
add_executable(dn_sequence_tests tests/sequence/test_sequence.cpp)
target_link_libraries(dn_sequence_tests PRIVATE dn_eval)
add_executable(dn_sequence_metric_tests tests/sequence/test_metrics.cpp)
target_link_libraries(dn_sequence_metric_tests PRIVATE dn_eval)
enable_testing()
add_test(NAME core COMMAND dn_core_tests)
add_test(NAME sequence COMMAND dn_sequence_tests)
add_test(NAME sequence_metrics COMMAND dn_sequence_metric_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  message(STATUS "Denoise sequence CLI test enabled with ${Python3_EXECUTABLE}")
  add_test(NAME sequence_cli COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/sequence/test_cli.py $<TARGET_FILE:dn_sequence_tool>)
else()
  message(STATUS "Denoise sequence CLI test NOT RUN: Python 3 interpreter unavailable")
endif()

# ---- Research prototype effect (CPU only; not part of releases) ---------------------------------
option(DN_BUILD_PLUGIN "Build the Premiere research prototype (requires Adobe SDKs)" OFF)
option(DN_BUILD_OFX "Build the OpenFX research prototype (downloads the BSD OpenFX SDK)" OFF)
set(VV_VERSION "0.1.0" CACHE STRING "Version written to bundles (CI sets it)")
add_executable(dn_effect_tests tests/core/test_effect.cpp src/effect/DnEffect.cpp src/effect/DnGpu.cpp)
target_include_directories(dn_effect_tests PRIVATE src/effect)
target_link_libraries(dn_effect_tests PRIVATE dn_eval)
add_test(NAME effect COMMAND dn_effect_tests)
add_executable(dn_gpu_emulation_tests tests/gpu/dn_gpu_emulation_test.cpp)
target_include_directories(dn_gpu_emulation_tests PRIVATE src/gpu/common)
target_link_libraries(dn_gpu_emulation_tests PRIVATE dn_eval)
add_test(NAME gpu_emulation COMMAND dn_gpu_emulation_tests)
option(DN_BUILD_GPU_TESTS "Build the Metal-vs-CPU comparison (macOS)" OFF)
# ---- GPU renderers: macOS Metal; Windows/Linux CUDA when a toolkit is found (or -DVV_WITH_CUDA=ON).
set(DN_EFFECT_SOURCES src/effect/DnEffect.cpp src/effect/DnGpu.cpp)
set(DN_EFFECT_INCLUDES src/effect src/core src/gpu/common src/gpu/metal src/gpu/cuda)
set(DN_GPU_LINK "")
set(DN_GPU_DEFS "")
set(DN_METALLIB "")
if(APPLE AND (DN_BUILD_PLUGIN OR DN_BUILD_OFX OR DN_BUILD_GPU_TESTS))
  enable_language(OBJCXX)
  include(${VV_ENGINE_DIR}/cmake/vv_metal.cmake)
  set(DN_METALLIB ${CMAKE_BINARY_DIR}/metal/Denoise.metallib)
  vv_add_metallib(dn_metallib SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/src/gpu/metal/Denoise.metal
                  OUTPUT ${DN_METALLIB} INCLUDES ${CMAKE_CURRENT_SOURCE_DIR}/src/core
                  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/dn_kernels.h ${CMAKE_CURRENT_SOURCE_DIR}/src/core/dn_color.h
                          ${VV_ENGINE_DIR}/gpu/metal/vv_common.metal ${VV_ENGINE_DIR}/core/vv/kernels.h)
  add_library(dn_metal_engine STATIC src/gpu/metal/DnMetalEngine.mm ${VV_ENGINE_DIR}/gpu/metal/vv_metal_backend.mm)
  target_include_directories(dn_metal_engine PUBLIC src/gpu/metal ${VV_ENGINE_DIR}/gpu/metal PRIVATE src/gpu/common)
  target_link_libraries(dn_metal_engine PUBLIC dn_core "-framework Foundation" "-framework Metal")
  target_compile_options(dn_metal_engine PRIVATE -fobjc-arc)
  set_target_properties(dn_metal_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
  list(APPEND DN_EFFECT_SOURCES src/effect/DnMetal.mm)
  list(APPEND DN_GPU_LINK dn_metal_engine)
  list(APPEND DN_GPU_DEFS DN_WITH_METAL_RENDERER)
elseif(NOT APPLE AND (DN_BUILD_PLUGIN OR DN_BUILD_OFX))
  include(CheckLanguage)
  check_language(CUDA)
  set(DN_CUDA_DEFAULT OFF)
  if(CMAKE_CUDA_COMPILER)
    set(DN_CUDA_DEFAULT ON)
  endif()
  if(DEFINED VV_WITH_CUDA)
    set(DN_CUDA_DEFAULT ${VV_WITH_CUDA})
  endif()
  option(DN_WITH_CUDA "Build the CUDA renderer (needs the CUDA toolkit)" ${DN_CUDA_DEFAULT})
  if(DN_WITH_CUDA)
    include(cmake/dn_cuda_engine.cmake)
    find_package(CUDAToolkit REQUIRED)
    list(APPEND DN_GPU_LINK dn_cuda_engine CUDA::cudart_static)
    list(APPEND DN_GPU_DEFS DN_WITH_CUDA_RENDERER)
  endif()
endif()

if(DN_BUILD_GPU_TESTS AND APPLE)
  add_executable(dn_metal_vs_cpu tests/gpu/dn_metal_vs_cpu.mm)
  target_link_libraries(dn_metal_vs_cpu PRIVATE dn_metal_engine dn_eval)
  target_compile_options(dn_metal_vs_cpu PRIVATE -fobjc-arc)
  add_dependencies(dn_metal_vs_cpu dn_metallib)
endif()

if(DN_BUILD_PLUGIN)
  include(${VV_ENGINE_DIR}/cmake/vv_premiere.cmake)
  set(DN_PR_ARGS)
  if(DN_METALLIB)
    list(APPEND DN_PR_ARGS GPU_METAL METALLIB ${DN_METALLIB} METALLIB_TARGET dn_metallib)
  endif()
  if(DN_WITH_CUDA)
    list(APPEND DN_PR_ARGS GPU_CUDA)
  endif()
  vv_add_premiere_effect(Denoise PIPL src/effect/DnPiPL.r BUNDLE_ID com.venostavisuals.Denoise
                         SOURCES ${DN_EFFECT_SOURCES} LINK dn_core ${DN_GPU_LINK} INCLUDES ${DN_EFFECT_INCLUDES}
                         ${DN_PR_ARGS})
  target_compile_definitions(Denoise PRIVATE ${DN_GPU_DEFS})
endif()
if(DN_BUILD_OFX)
  include(${VV_ENGINE_DIR}/cmake/vv_ofx.cmake)
  # CPU only on purpose: the GPU renderers are spatial only, and the OpenFX GPU path has no CPU
  # fallback, so a GPU build would lose temporal fusion in Resolve.
  vv_add_ofx_bundle(DenoiseOFX NAME Denoise DISPLAY_NAME "Denoise (research prototype)"
                    BUNDLE_ID com.venostavisuals.Denoise.ofx VERSION ${VV_VERSION}
                    SOURCES src/effect/DnEffect.cpp src/effect/DnGpu.cpp LINK dn_core
                    EXPECT_ID com.venostavisuals.Denoise)
  target_include_directories(DenoiseOFX PRIVATE ${DN_EFFECT_INCLUDES})
endif()
