cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED CMAKE_OSX_ARCHITECTURES)
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
project(Retro VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(NOT TARGET vv_core)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../shared/engine ${CMAKE_BINARY_DIR}/vv-engine)
endif()

add_library(rt_core STATIC src/core/rt.cpp)
target_include_directories(rt_core PUBLIC src/core)
target_link_libraries(rt_core PUBLIC vv_core)
set_target_properties(rt_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
  target_compile_options(rt_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(RT_BUILD_PLUGIN "Build the native Premiere plugin (requires Adobe SDKs)" OFF)
option(RT_BUILD_OFX "Build the OpenFX plug-in bundle (downloads the BSD OpenFX SDK)" OFF)
option(RT_BUILD_GPU_TESTS "Build the Metal-vs-CPU comparison (macOS)" OFF)
set(VV_VERSION "${PROJECT_VERSION}" CACHE STRING "Version written to bundles (CI sets it)")

add_executable(rt_core_tests tests/core/test_main.cpp src/effect/RtEffect.cpp src/effect/RtCudaRenderer.cpp)
target_include_directories(rt_core_tests PRIVATE src/effect src/gpu/common)
target_link_libraries(rt_core_tests PRIVATE rt_core)
target_compile_definitions(rt_core_tests PRIVATE RT_NO_GPU_RENDERERS)
add_executable(rt_gpu_emulation_tests tests/gpu/rt_gpu_emulation_test.cpp)
target_include_directories(rt_gpu_emulation_tests PRIVATE src/gpu/common)
target_link_libraries(rt_gpu_emulation_tests PRIVATE rt_core)
enable_testing()
add_test(NAME core COMMAND rt_core_tests)
add_test(NAME gpu_emulation COMMAND rt_gpu_emulation_tests)

# ---- GPU renderers ------------------------------------------------------------------------------
# macOS: Metal (always). Windows/Linux: CUDA when a CUDA toolkit is found (or -DVV_WITH_CUDA=ON).
set(RT_EFFECT_SOURCES src/effect/RtEffect.cpp src/effect/RtCudaRenderer.cpp)
set(RT_EFFECT_INCLUDES src/effect src/core src/gpu/common src/gpu/metal src/gpu/cuda)
set(RT_GPU_LINK "")
set(RT_GPU_DEFS "")
set(RT_METALLIB "")
if(APPLE AND (RT_BUILD_PLUGIN OR RT_BUILD_OFX OR RT_BUILD_GPU_TESTS))
  enable_language(OBJCXX)
  include(${VV_ENGINE_DIR}/cmake/vv_metal.cmake)
  set(RT_METALLIB ${CMAKE_BINARY_DIR}/metal/Retro.metallib)
  vv_add_metallib(rt_metallib SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/src/gpu/metal/Retro.metal
                  OUTPUT ${RT_METALLIB} INCLUDES ${CMAKE_CURRENT_SOURCE_DIR}/src/core
                  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/rt_kernels.h ${VV_ENGINE_DIR}/gpu/metal/vv_common.metal
                          ${VV_ENGINE_DIR}/core/vv/kernels.h)
  add_library(rt_metal_engine STATIC src/gpu/metal/RetroMetalEngine.mm ${VV_ENGINE_DIR}/gpu/metal/vv_metal_backend.mm)
  target_include_directories(rt_metal_engine PUBLIC src/gpu/metal ${VV_ENGINE_DIR}/gpu/metal PRIVATE src/gpu/common)
  target_link_libraries(rt_metal_engine PUBLIC rt_core "-framework Foundation" "-framework Metal")
  target_compile_options(rt_metal_engine PRIVATE -fobjc-arc)
  set_target_properties(rt_metal_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
  list(APPEND RT_EFFECT_SOURCES src/effect/RtMetalRenderer.mm)
  list(APPEND RT_GPU_LINK rt_metal_engine)
elseif(NOT APPLE AND (RT_BUILD_PLUGIN OR RT_BUILD_OFX))
  include(CheckLanguage)
  check_language(CUDA)
  set(RT_CUDA_DEFAULT OFF)
  if(CMAKE_CUDA_COMPILER)
    set(RT_CUDA_DEFAULT ON)
  endif()
  if(DEFINED VV_WITH_CUDA)  # set by the shared CI workflows
    set(RT_CUDA_DEFAULT ${VV_WITH_CUDA})
  endif()
  option(RT_WITH_CUDA "Build the CUDA renderer (needs the CUDA toolkit)" ${RT_CUDA_DEFAULT})
  if(RT_WITH_CUDA)
    include(cmake/rt_cuda_engine.cmake)
    find_package(CUDAToolkit REQUIRED)
    list(APPEND RT_GPU_LINK rt_cuda_engine CUDA::cudart_static)
    list(APPEND RT_GPU_DEFS RT_WITH_CUDA_RENDERER)
    message(STATUS "Retro: CUDA renderer enabled")
  endif()
endif()

# Metal engine on a Mac GPU vs the CPU reference (CI: retro-metal).
if(RT_BUILD_GPU_TESTS AND APPLE)
  add_executable(rt_metal_vs_cpu tests/gpu/rt_metal_vs_cpu.mm)
  target_link_libraries(rt_metal_vs_cpu PRIVATE rt_metal_engine)
  target_compile_options(rt_metal_vs_cpu PRIVATE -fobjc-arc)
  add_dependencies(rt_metal_vs_cpu rt_metallib)
endif()

if(RT_BUILD_PLUGIN)
  include(${VV_ENGINE_DIR}/cmake/vv_premiere.cmake)
  set(RT_PR_ARGS)
  if(RT_METALLIB)
    list(APPEND RT_PR_ARGS GPU_METAL METALLIB ${RT_METALLIB} METALLIB_TARGET rt_metallib)
  endif()
  if(RT_WITH_CUDA)
    list(APPEND RT_PR_ARGS GPU_CUDA)
  endif()
  vv_add_premiere_effect(Retro PIPL src/effect/RtPiPL.r BUNDLE_ID com.venostavisuals.Retro
                         SOURCES ${RT_EFFECT_SOURCES} LINK rt_core ${RT_GPU_LINK} INCLUDES ${RT_EFFECT_INCLUDES}
                         ${RT_PR_ARGS})
  target_compile_definitions(Retro PRIVATE ${RT_GPU_DEFS})
endif()

if(RT_BUILD_OFX)
  include(${VV_ENGINE_DIR}/cmake/vv_ofx.cmake)
  set(RT_OFX_METAL)
  if(RT_METALLIB)
    set(RT_OFX_METAL METALLIB ${RT_METALLIB} METALLIB_TARGET rt_metallib)
  endif()
  vv_add_ofx_bundle(RetroOFX NAME Retro DISPLAY_NAME "Retro"
                    BUNDLE_ID com.venostavisuals.Retro.ofx VERSION ${VV_VERSION}
                    SOURCES ${RT_EFFECT_SOURCES} LINK rt_core ${RT_GPU_LINK}
                    EXPECT_ID com.venostavisuals.Retro ${RT_OFX_METAL})
  target_include_directories(RetroOFX PRIVATE ${RT_EFFECT_INCLUDES})
  target_compile_definitions(RetroOFX PRIVATE ${RT_GPU_DEFS})
endif()
