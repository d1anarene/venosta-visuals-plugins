cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED CMAKE_OSX_ARCHITECTURES)
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
project(ExposureFalseColor VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(NOT TARGET vv_core)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../shared/engine ${CMAKE_BINARY_DIR}/vv-engine)
endif()

add_library(ef_core STATIC src/core/ef.cpp)
target_include_directories(ef_core PUBLIC src/core)
target_link_libraries(ef_core PUBLIC vv_core)
set_target_properties(ef_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
  target_compile_options(ef_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(EF_BUILD_PLUGIN "Build the native Premiere plugin (requires Adobe SDKs)" OFF)
option(EF_BUILD_OFX "Build the OpenFX plug-in bundle (downloads the BSD OpenFX SDK)" OFF)
option(EF_BUILD_GPU_TESTS "Build the Metal-vs-CPU comparison (macOS)" OFF)
set(VV_VERSION "${PROJECT_VERSION}" CACHE STRING "Version written to bundles (CI sets it)")

add_executable(ef_core_tests tests/core/test_main.cpp src/effect/EfEffect.cpp src/effect/EfCudaRenderer.cpp)
target_include_directories(ef_core_tests PRIVATE src/effect src/gpu/common)
target_link_libraries(ef_core_tests PRIVATE ef_core)
target_compile_definitions(ef_core_tests PRIVATE EF_NO_GPU_RENDERERS)
enable_testing()
add_test(NAME core COMMAND ef_core_tests)

# ---- GPU renderers ------------------------------------------------------------------------------
set(EF_EFFECT_SOURCES src/effect/EfEffect.cpp src/effect/EfCudaRenderer.cpp)
set(EF_EFFECT_INCLUDES src/effect src/core src/gpu/common src/gpu/metal src/gpu/cuda)
set(EF_GPU_LINK "")
set(EF_GPU_DEFS "")
set(EF_METALLIB "")
if(APPLE AND (EF_BUILD_PLUGIN OR EF_BUILD_OFX OR EF_BUILD_GPU_TESTS))
  enable_language(OBJCXX)
  include(${VV_ENGINE_DIR}/cmake/vv_metal.cmake)
  set(EF_METALLIB ${CMAKE_BINARY_DIR}/metal/ExposureFalseColor.metallib)
  vv_add_metallib(ef_metallib SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/src/gpu/metal/ExposureFalseColor.metal
                  OUTPUT ${EF_METALLIB} INCLUDES ${CMAKE_CURRENT_SOURCE_DIR}/src/core
                  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/core/ef_kernels.h ${VV_ENGINE_DIR}/gpu/metal/vv_common.metal)
  add_library(ef_metal_engine STATIC src/gpu/metal/EfMetalEngine.mm ${VV_ENGINE_DIR}/gpu/metal/vv_metal_backend.mm)
  target_include_directories(ef_metal_engine PUBLIC src/gpu/metal ${VV_ENGINE_DIR}/gpu/metal)
  target_link_libraries(ef_metal_engine PUBLIC ef_core "-framework Foundation" "-framework Metal")
  target_compile_options(ef_metal_engine PRIVATE -fobjc-arc)
  set_target_properties(ef_metal_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
  list(APPEND EF_EFFECT_SOURCES src/effect/EfMetalRenderer.mm)
  list(APPEND EF_GPU_LINK ef_metal_engine)
elseif(NOT APPLE AND (EF_BUILD_PLUGIN OR EF_BUILD_OFX))
  include(CheckLanguage)
  check_language(CUDA)
  set(EF_CUDA_DEFAULT OFF)
  if(CMAKE_CUDA_COMPILER)
    set(EF_CUDA_DEFAULT ON)
  endif()
  if(DEFINED VV_WITH_CUDA)
    set(EF_CUDA_DEFAULT ${VV_WITH_CUDA})
  endif()
  option(EF_WITH_CUDA "Build the CUDA renderer (needs the CUDA toolkit)" ${EF_CUDA_DEFAULT})
  if(EF_WITH_CUDA)
    include(cmake/ef_cuda_engine.cmake)
    find_package(CUDAToolkit REQUIRED)
    list(APPEND EF_GPU_LINK ef_cuda_engine CUDA::cudart_static)
    list(APPEND EF_GPU_DEFS EF_WITH_CUDA_RENDERER)
  endif()
endif()

if(EF_BUILD_GPU_TESTS AND APPLE)
  add_executable(ef_metal_vs_cpu tests/gpu/ef_metal_vs_cpu.mm)
  target_link_libraries(ef_metal_vs_cpu PRIVATE ef_metal_engine)
  target_compile_options(ef_metal_vs_cpu PRIVATE -fobjc-arc)
  add_dependencies(ef_metal_vs_cpu ef_metallib)
endif()

if(EF_BUILD_PLUGIN)
  include(${VV_ENGINE_DIR}/cmake/vv_premiere.cmake)
  set(EF_PR_ARGS)
  if(EF_METALLIB)
    list(APPEND EF_PR_ARGS GPU_METAL METALLIB ${EF_METALLIB} METALLIB_TARGET ef_metallib)
  endif()
  if(EF_WITH_CUDA)
    list(APPEND EF_PR_ARGS GPU_CUDA)
  endif()
  vv_add_premiere_effect(ExposureFalseColor PIPL src/effect/EfPiPL.r BUNDLE_ID com.venostavisuals.ExposureFalseColor
                         SOURCES ${EF_EFFECT_SOURCES} LINK ef_core ${EF_GPU_LINK} INCLUDES ${EF_EFFECT_INCLUDES}
                         ${EF_PR_ARGS})
  target_compile_definitions(ExposureFalseColor PRIVATE ${EF_GPU_DEFS})
endif()

if(EF_BUILD_OFX)
  include(${VV_ENGINE_DIR}/cmake/vv_ofx.cmake)
  set(EF_OFX_METAL)
  if(EF_METALLIB)
    set(EF_OFX_METAL METALLIB ${EF_METALLIB} METALLIB_TARGET ef_metallib)
  endif()
  vv_add_ofx_bundle(ExposureFalseColorOFX NAME ExposureFalseColor DISPLAY_NAME "Exposure &amp; False Color"
                    BUNDLE_ID com.venostavisuals.ExposureFalseColor.ofx VERSION ${VV_VERSION}
                    SOURCES ${EF_EFFECT_SOURCES} LINK ef_core ${EF_GPU_LINK}
                    EXPECT_ID com.venostavisuals.ExposureFalseColor ${EF_OFX_METAL})
  target_include_directories(ExposureFalseColorOFX PRIVATE ${EF_EFFECT_INCLUDES})
  target_compile_definitions(ExposureFalseColorOFX PRIVATE ${EF_GPU_DEFS})
endif()
