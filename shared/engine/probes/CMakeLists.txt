if(VV_PROBE_PREMIERE OR VV_PROBE_METAL OR VV_PROBE_CUDA)

  function(add_probe target)
    set(sources ${ARGN})
    if(NOT sources)
      set(sources premiere_cpu_probe.cpp)
    endif()
    add_library(${target} MODULE ${sources} ${AE_SDK_UTIL}/AEFX_SuiteHelper.c)
    target_link_libraries(${target} PRIVATE vv_core)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${VV_PREMIERE_DIR}
      ${AE_SDK_HEADERS} ${AE_SDK_HEADERS}/SP ${AE_SDK_HEADERS}/adobesdk ${AE_SDK_UTIL}
      ${PR_SDK_HEADERS} ${PR_SDK_UTILS} ${PR_GPU_UTILS})
    if(APPLE)
      target_link_libraries(${target} PRIVATE "-framework Foundation")
      set_target_properties(${target} PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin
        MACOSX_BUNDLE_INFO_PLIST ${VV_PREMIERE_DIR}/Info.plist.in
        MACOSX_BUNDLE_GUI_IDENTIFIER com.venostavisuals.research.${target}
        MACOSX_BUNDLE_BUNDLE_NAME ${target})
      set(rsrc ${CMAKE_CURRENT_BINARY_DIR}/${target}.rsrc)
      add_custom_command(OUTPUT ${rsrc}
        COMMAND xcrun Rez -o ${rsrc} -define __MACH__ -define AE_OS_MAC -useDF -arch arm64 -arch x86_64
          -i ${AE_SDK_HEADERS} -i ${AE_SDK_HEADERS}/SP -i ${AE_SDK_RESOURCES} -i ${CMAKE_CURRENT_SOURCE_DIR}
          ${CMAKE_CURRENT_SOURCE_DIR}/${target}.r
        DEPENDS ${target}.r probe_pipl.r VERBATIM)
      target_sources(${target} PRIVATE ${rsrc})
      set_source_files_properties(${rsrc} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    else()
      set_target_properties(${target} PROPERTIES SUFFIX ".aex" PREFIX "")
      target_compile_definitions(${target} PRIVATE MSWindows WIN32 _WINDOWS NOMINMAX)
      set(gen ${CMAKE_CURRENT_BINARY_DIR}/${target}-pipl)
      file(MAKE_DIRECTORY ${gen})
      add_custom_command(OUTPUT ${gen}/${target}.rc
        COMMAND cl /nologo /EP /P /Fi${gen}/${target}.rr /TC /I${AE_SDK_HEADERS} /I${AE_SDK_HEADERS}/SP
          /I${AE_SDK_RESOURCES} /I${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/${target}.r
        COMMAND ${AE_SDK_RESOURCES}/PiPLtool.exe ${gen}/${target}.rr ${gen}/${target}.rrc
        COMMAND cl /nologo /EP /P /Fi${gen}/${target}.rc /DMSWindows /TC ${gen}/${target}.rrc
        DEPENDS ${target}.r probe_pipl.r WORKING_DIRECTORY ${gen} VERBATIM)
      target_sources(${target} PRIVATE ${gen}/${target}.rc)
    endif()
  endfunction()
